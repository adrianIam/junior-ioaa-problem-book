\begin{description}
\item[(a)] Draw the spherical triangle North Celestial Pole--Zenith--Star.
\ioaafig{2025_TH_Q06-s1.pdf}
The spherical law of cosines gives
\[
\sin a_{\mathrm{Altair}}=\sin\varphi\sin\delta_{\mathrm{Altair}}+\cos\varphi\cos\delta_{\mathrm{Altair}}\cos H_{\mathrm{Altair}}.
\]
At rising, neglecting horizon effects and refraction, $a_{\mathrm{Altair}}=0^\circ$. Hence
\[
\cos H_{\mathrm{Altair}}=-\tan\varphi\tan\delta_{\mathrm{Altair}},\qquad
H_{\mathrm{Altair}}=\pm6^{\mathrm h}31^{\mathrm m}.
\]\hfill(2 points)
Altair rises $6^{\mathrm h}31^{\mathrm m}$ before crossing the meridian. \hfill(1 point)
On the meridian $T_{\mathrm{sidereal}}=\alpha_{\mathrm{Altair}}=19^{\mathrm h}51^{\mathrm m}$. \hfill(1 point)
Thus $T_{\mathrm{sidereal}}=19^{\mathrm h}51^{\mathrm m}-6^{\mathrm h}31^{\mathrm m}=13^{\mathrm h}20^{\mathrm m}$. \hfill(1 point)
\item[(b)] At rising, $T_{\mathrm{local}}=22{:}39{:}43$ and $H_{\mathrm{Sun}}=10{:}39{:}43$. \hfill(1 point)
The sidereal time is $H_\gamma=13^{\mathrm h}20^{\mathrm m}$. \hfill(0.5 point)
Therefore $\alpha_{\mathrm{Sun}}=2^{\mathrm h}40^{\mathrm m}17^{\mathrm s}$. \hfill(1 point)
This is 40.75 days after the spring equinox, \hfill(0.5 point)
so the date is 1 May. \hfill(1 point)
\item[(c)] The Sun's right ascension changes negligibly in a few hours, so $\Delta T_{\mathrm{local}}\approx\Delta T_{\mathrm{sidereal}}$. When the stars have the same altitude,
\begin{align*}
T_{\mathrm{sidereal},2}&\approx T_{\mathrm{sidereal},1}+T_{\mathrm{local},2}-T_{\mathrm{local},1}\\
&=13^{\mathrm h}20^{\mathrm m}+2^{\mathrm h}43^{\mathrm m}-22^{\mathrm h}40^{\mathrm m}=17^{\mathrm h}23^{\mathrm m}.
\end{align*}\hfill(1 point)
For each star,
\begin{align*}
\sin a_{\mathrm{Altair}}&=\sin\varphi\sin\delta_{\mathrm{Altair}}+\cos\varphi\cos\delta_{\mathrm{Altair}}\cos H_{\mathrm{Altair}},\\
\sin a_{\mathrm{Arcturus}}&=\sin\varphi\sin\delta_{\mathrm{Arcturus}}+\cos\varphi\cos\delta_{\mathrm{Arcturus}}\cos H_{\mathrm{Arcturus}}.
\end{align*}\hfill(1 point)
Equating the two altitudes gives
\[
\cos H_{\mathrm{Arcturus}}=
\frac{\sin\varphi\sin\delta_{\mathrm{Altair}}+\cos\varphi\cos\delta_{\mathrm{Altair}}\cos H'_{\mathrm{Altair}}-\sin\varphi\sin\delta_{\mathrm{Arcturus}}}{\cos\varphi\cos\delta_{\mathrm{Arcturus}}}.
\]
The key gives $H_{\mathrm{Arcturus}}\in\{3^{\mathrm h}10^{\mathrm m}36^{\mathrm s},20^{\mathrm h}49^{\mathrm m}30^{\mathrm s}\}$. The western solution is $3^{\mathrm h}10^{\mathrm m}36^{\mathrm s}$. \hfill(3 points)
Accept $3^{\mathrm h}9^{\mathrm m}$--$3^{\mathrm h}12^{\mathrm m}$. Now
\[
\alpha_{\mathrm{Arcturus}}=T_{\mathrm{sidereal},2}-H_{\mathrm{Arcturus}}\approx14^{\mathrm h}12^{\mathrm m}.
\]\hfill(1 point)
Accept $14^{\mathrm h}10^{\mathrm m}$--$14^{\mathrm h}18^{\mathrm m}$.
\end{description}
